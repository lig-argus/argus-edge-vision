import os
from pathlib import Path
import select
import subprocess
import sys

import pytest
import zmq

from argus_vision.bus import Publisher, Subscriber


def test_real_bus_forwards_multipart_and_exits_on_sigterm():
    root = Path(__file__).parents[1]
    env = dict(os.environ, PYTHONPATH=str(root / "src"), PYTHONDONTWRITEBYTECODE="1")
    process = subprocess.Popen([sys.executable, "-m", "argus_vision.busd",
                                "--publish-bind", "tcp://127.0.0.1:*",
                                "--subscribe-bind", "tcp://127.0.0.1:*"],
                               stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, env=env)
    try:
        assert select.select([process.stdout], [], [], 5)[0], "bus did not become ready"
        line = process.stdout.readline().strip()
        assert "ARGUS bus:" in line, line
        endpoints = line.split("publishers -> ")[1]
        publish, subscribe = endpoints.split(", subscribers <- ")
        with Publisher(publish) as publisher, Subscriber(["probe"], subscribe) as subscriber:
            received = None
            for sequence in range(20):
                publisher.publish("probe", {"sequence": sequence}, b"payload")
                try:
                    received = subscriber.receive(timeout_ms=100)
                    break
                except TimeoutError:
                    pass
            assert received is not None
            assert received[0] == "probe" and received[2] == b"payload"
        process.terminate()
        assert process.wait(timeout=2) == 0
    finally:
        if process.poll() is None:
            process.kill()
        process.communicate(timeout=2)
