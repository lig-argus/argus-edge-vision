import numpy as np

from argus_vision.frame_queue import CapturedFrame, LatestFrameQueue


def make_frame(frame_id: int) -> CapturedFrame:
    return CapturedFrame(frame_id, np.zeros((1, 1, 3), dtype=np.uint8), 0, 0)


def test_queue_discards_oldest_frame_when_full():
    frames = LatestFrameQueue(capacity=2)
    frames.put(make_frame(1))
    frames.put(make_frame(2))
    frames.put(make_frame(3))
    assert frames.dropped_frames == 1
    assert frames.get().frame_id == 2
    assert frames.get().frame_id == 3
