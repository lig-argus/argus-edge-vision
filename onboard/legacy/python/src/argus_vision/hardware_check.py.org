from __future__ import annotations

import argparse
import platform
import shutil
import subprocess
import sys
from pathlib import Path

from .clock import SystemClock
from .frame_sources import FrameSourceOptions, OpenCvFrameSource, Raw14SdkFrameSource


def pass_check(name: str, detail: str) -> None:
    print(f"[PASS] {name}: {detail}")


def fail_check(name: str, detail: str) -> None:
    print(f"[FAIL] {name}: {detail}", file=sys.stderr)


def check_hailo_device() -> bool:
    executable = shutil.which("hailortcli")
    if executable is None:
        fail_check("HailoRT", "hailortcli not found; install the Raspberry Pi hailo-all package")
        return False
    version_result = subprocess.run(
        [executable, "--version"],
        capture_output=True,
        text=True,
        timeout=10,
        check=False,
    )
    version_output = (version_result.stdout + version_result.stderr).strip()
    if version_result.returncode != 0 or "4.23." not in version_output:
        fail_check(
            "HailoRT",
            f"Hailo-8 runtime 4.23.x is required; detected: {version_output or 'unknown'}",
        )
        return False
    pass_check("HailoRT", version_output)

    result = subprocess.run(
        [executable, "fw-control", "identify"],
        capture_output=True,
        text=True,
        timeout=15,
        check=False,
    )
    output = (result.stdout + result.stderr).strip()
    if result.returncode != 0:
        fail_check("Hailo-8", output or f"hailortcli exited with {result.returncode}")
        return False
    normalized = output.upper().replace("-", "")
    if "HAILO8L" in normalized:
        fail_check("Hailo-8", "a Hailo-8L was detected, but this profile requires Hailo-8")
        return False
    if "HAILO8" not in normalized:
        fail_check("Hailo-8", "device responded, but its architecture was not identified as Hailo-8")
        return False
    pass_check("Hailo-8", "Hailo-8 device identified by hailortcli")
    return True


def check_python_binding() -> bool:
    try:
        import hailo_platform  # noqa: F401
    except Exception as exc:
        fail_check("PyHailoRT", f"cannot import hailo_platform: {exc}")
        return False
    pass_check("PyHailoRT", "hailo_platform import succeeded")
    return True


def check_hef(path_value: str | None, labels_value: str | None) -> bool:
    if not path_value:
        print("[SKIP] HEF: no model path supplied")
        return True
    path = Path(path_value)
    if not path.is_file():
        fail_check("HEF", f"file not found: {path}")
        return False
    try:
        from hailo_platform import HEF

        hef = HEF(str(path))
        inputs = hef.get_input_vstream_infos()
        outputs = hef.get_output_vstream_infos()
    except Exception as exc:
        fail_check("HEF", f"cannot parse {path.name}: {exc}")
        return False
    if len(inputs) != 1:
        fail_check("HEF", f"expected one input, got {len(inputs)}")
        return False
    if tuple(inputs[0].shape)[-1] != 3:
        fail_check("HEF", f"expected RGB NHWC input, got {inputs[0].shape}")
        return False
    input_type = str(inputs[0].format.type).upper()
    if "UINT8" not in input_type:
        fail_check("HEF", f"expected UINT8 input, got {inputs[0].format.type}")
        return False
    if len(outputs) != 1:
        fail_check(
            "HEF",
            f"expected one HailoRT NMS output, got {len(outputs)}; compile YOLOX with NMS",
        )
        return False
    output_order = str(outputs[0].format.order).upper()
    if "NMS" not in output_order:
        fail_check(
            "HEF",
            f"output is not HailoRT NMS ({outputs[0].format.order}); this runtime needs an NMS HEF",
        )
        return False
    output_shape = tuple(outputs[0].shape)
    nms_classes = int(outputs[0].nms_shape.number_of_classes)
    if labels_value:
        labels_path = Path(labels_value)
        if not labels_path.is_file():
            fail_check("labels", f"file not found: {labels_path}")
            return False
        labels = [
            line.strip()
            for line in labels_path.read_text(encoding="utf-8").splitlines()
            if line.strip() and not line.lstrip().startswith("#")
        ]
        if nms_classes != len(labels):
            fail_check(
                "labels",
                f"HEF classes={nms_classes}, labels={len(labels)}",
            )
            return False
        pass_check("labels", f"{len(labels)} classes match the HEF output")
    pass_check(
        "HEF",
        f"{path.name}, input={inputs[0].shape}, output={outputs[0].name} {output_shape}, format={output_order}",
    )
    return True


def check_frame_source(args: argparse.Namespace) -> bool:
    options = FrameSourceOptions(camera=args.camera, width=args.width, height=args.height,
                                 fourcc=args.fourcc, ir_capture=args.ir_capture,
                                 ir_device=args.ir_device, ir_timeout_s=args.ir_timeout)
    source = None
    try:
        if args.frame_source == "ir_sdk":
            source = Raw14SdkFrameSource(options, SystemClock())
        elif args.frame_source in ("ir_v4l2", "replay"):
            source = OpenCvFrameSource(options, SystemClock(), replay=args.frame_source == "replay")
        else:
            raise NotImplementedError(f"input adapter is unavailable: {args.frame_source}")
        ok, frame = source.read()
        if not ok or frame is None:
            raise RuntimeError("source produced no valid frame")
        pass_check("frame source", f"{args.frame_source}, shape={frame.image.shape}; sensor exposure time unavailable")
        return True
    except Exception as exc:
        fail_check("frame source", str(exc))
        return False
    finally:
        if source is not None:
            source.release()


def main() -> None:
    parser = argparse.ArgumentParser(description="Check Raspberry Pi, Hailo-8, RGB camera and YOLOX HEF")
    parser.add_argument("--camera", default="/dev/video0")
    parser.add_argument("--width", type=int, default=640)
    parser.add_argument("--height", type=int, default=480)
    parser.add_argument("--fourcc", default="MJPG")
    parser.add_argument("--hef")
    parser.add_argument("--labels")
    parser.add_argument("--frame-source", choices=("ir_v4l2", "ir_sdk", "replay", "gazebo_camera", "gazebo_pose"), default="ir_v4l2")
    parser.add_argument("--ir-capture", default="/home/user/thessen_raw14_viewer/build/i3_raw14_capture")
    parser.add_argument("--ir-device", type=int, default=0)
    parser.add_argument("--ir-timeout", type=float, default=15)
    args = parser.parse_args()

    print(f"Platform: {platform.platform()} ({platform.machine()})")
    results = [
        check_hailo_device(),
        check_python_binding(),
        check_frame_source(args),
        check_hef(args.hef, args.labels),
    ]
    if not all(results):
        raise SystemExit(1)
    print("All required hardware checks passed.")


if __name__ == "__main__":
    main()
