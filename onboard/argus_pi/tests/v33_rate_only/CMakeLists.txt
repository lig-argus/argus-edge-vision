cmake_minimum_required(VERSION 3.16)
project(v33_rate_only_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ARGUS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CANDIDATE "${ARGUS_ROOT}")
add_library(rate_core STATIC
 ${CANDIDATE}/src/argus/control/guidance/predictive_follower.cpp
 ${ARGUS_ROOT}/src/argus/control/geometry/calibrated_ground_geometry.cpp)
target_include_directories(rate_core PUBLIC ${CANDIDATE}/src ${ARGUS_ROOT}/src ${ARGUS_ROOT}/vendor/vmy ${ARGUS_ROOT}/vendor/vmy/eigen ${ARGUS_ROOT}/vendor/yaw/msgpack/include)
if(MSVC)
 target_compile_options(rate_core PUBLIC /utf-8 /UNDEBUG)
else()
 target_compile_options(rate_core PUBLIC -UNDEBUG)
endif()
enable_testing()
add_executable(rate_only_test rate_only_test.cpp)
target_link_libraries(rate_only_test PRIVATE rate_core)
add_test(NAME v33_rate_only_commit_safety COMMAND rate_only_test)
add_executable(retained_vertical_test ${ARGUS_ROOT}/tests/follow_b/predictive_vertical_slew_test.cpp)
target_link_libraries(retained_vertical_test PRIVATE rate_core)
add_test(NAME retained_vertical COMMAND retained_vertical_test)
add_executable(retained_horizontal_test ${ARGUS_ROOT}/tests/follow_b/v32_sent_command_slew_test.cpp)
target_link_libraries(retained_horizontal_test PRIVATE rate_core)
add_test(NAME retained_horizontal COMMAND retained_horizontal_test)
