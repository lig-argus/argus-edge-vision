cmake_minimum_required(VERSION 3.16)
project(argus_pi_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
add_library(argus_warnings INTERFACE)
target_compile_options(argus_warnings INTERFACE -Wall -Wextra -Wpedantic -Wshadow)
set(BUILD_TESTING OFF CACHE BOOL "Runtime package excludes test fixtures" FORCE)
include(cmake/perception.cmake)
option(ARGUS_BUILD_CAMERA "Build I3 managed camera helper" ON)
if(ARGUS_BUILD_CAMERA)
 include(cmake/camera_i3.cmake)
endif()
